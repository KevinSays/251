#!/usr/bin/env bash
# IT251 - persistent terminal history logging setup. Run once.
set -euo pipefail

MARKER_START="# >>> IT251 history logging >>>"
MARKER_END="# <<< IT251 history logging <<<"
LOG_DIR="$HOME/IT251/logs"

case "$SHELL" in
  */zsh) RC_FILE="$HOME/.zshrc" ;;
  *)     RC_FILE="$HOME/.bashrc" ;;
esac

mkdir -p "$LOG_DIR"

if grep -qF "$MARKER_START" "$RC_FILE" 2>/dev/null; then
  echo "Logging block already present in $RC_FILE - nothing to do."
else
  {
    echo ""
    echo "$MARKER_START"
    if [[ "$RC_FILE" == *zshrc ]]; then
      cat <<'ZSH'
export HISTFILE="$HOME/.zsh_history"
HISTSIZE=100000
SAVEHIST=100000
setopt INC_APPEND_HISTORY
setopt EXTENDED_HISTORY
ZSH
    else
      cat <<'BASH'
export HISTFILE="$HOME/.bash_history"
export HISTTIMEFORMAT="%F %T  "
export HISTSIZE=100000
export HISTFILESIZE=100000
shopt -s histappend
PROMPT_COMMAND="history -a; history -c; history -r${PROMPT_COMMAND:+; $PROMPT_COMMAND}"
BASH
    fi
    echo "$MARKER_END"
  } >> "$RC_FILE"
  echo "Added logging config to $RC_FILE."
fi

echo "Done. Run: source $RC_FILE   (or open a new terminal) then check with: history | tail"
